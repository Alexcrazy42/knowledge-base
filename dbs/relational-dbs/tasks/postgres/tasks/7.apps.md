## 7.3. Упражнение

Напишите запрос, для каждого города показывающий кол-во пассажиров, прилетающих в него из Москвы в какой-нибудь определенный день. Запрос должен выдавать названия населенных пунктов на английском языке.

```sql
select ad.city->>'en', count(*)
from tickets t 
join segments s on s.ticket_no = t.ticket_no 
join flights f on f.flight_id = s.flight_id 
join routes r on r.route_no = f.route_no 
join airports_data ad on ad.airport_code = r.arrival_airport 
where r.departure_airport in ('BKA', 'DME', 'OSF', 'SVO', 'VKO')
and f.scheduled_departure >= '2025-10-01' 
  AND f.scheduled_departure < '2025-10-02'
group by ad.city->>'en'
```

## 7.4. Упражнение

Реализуйте функцию покупки билета. С точки зрения пассажира покупка состоит из двух этапов. На первом этапе система выдает список возможных перелетов для указанных параметров: пункты отправления и назначения, кол-во мест, дата вылета. На втором - выбранный пассажиром вариант оформляется в виде бронирования, и пассажиру возвращается номер бронирования.

Поиск:

```sql
select * from routes r
join flights f on r.route_no = f.route_no
where r.departure_airport = 'SVO' and r.arrival_airport = 'KZN' 
and f.scheduled_departure >= '2025-10-05' 
AND f.scheduled_departure < '2025-10-06'
```

Бронирование:

```sql
CREATE SEQUENCE bookings.ticket_no_seq
	INCREMENT BY 1
	MINVALUE 1
	MAXVALUE 9999999999999
	START 5434990768
	CACHE 1
	NO CYCLE;


select nextval('bookings.ticket_no_seq');


CREATE OR REPLACE FUNCTION bookings.generate_tocket_no()
RETURNS bpchar
LANGUAGE sql
AS $$
    SELECT LPAD(nextval('bookings.ticket_no_seq')::text, 13, '0')::bpchar;
$$;

create type airplane_fare_info as (
	v_flight_id int4,
	airplane_code bpchar(3),
	fare_condition text
)

CREATE TYPE passenger_info AS (
    passenger_name text,
   	airplane_fare_infos airplane_fare_info[]
);


create function create_booking(passengers passenger_info[]) returns character
    language plpgsql
as
$$
DECLARE
    v_book_ref bpchar(6);
    v_ticket_no TEXT;
    v_passenger_id TEXT;
    v_total_amount numeric(10, 2);
    v_passenger passenger_info;
    v_airplane_fare_info airplane_fare_info;
BEGIN


    v_book_ref := bookings.generate_book_ref();
    v_total_amount := 0;
    INSERT INTO bookings
    (book_ref, book_date, total_amount)
    VALUES
        (v_book_ref, CURRENT_DATE, v_total_amount);


    FOREACH v_passenger IN ARRAY passengers
        LOOP
            v_ticket_no := bookings.generate_tocket_no();
            v_passenger_id := bookings.generate_passenger_id();

            -- вставляем билеты
            INSERT INTO tickets
            (ticket_no, book_ref, passenger_id, passenger_name, outbound)
            VALUES
                (v_ticket_no, v_book_ref, v_passenger_id, v_passenger.passenger_name, false);


            FOREACH v_airplane_fare_info IN ARRAY v_passenger.airplane_fare_infos
                LOOP

                    -- для каждого билета вставляем перелеты
                    INSERT INTO segments
                    (ticket_no, flight_id, fare_conditions, price)
                    VALUES
                        (v_ticket_no, v_airplane_fare_info.v_flight_id, v_airplane_fare_info.fare_condition, 1000); -- нигде нет инфы о ценнике, поэтому вставляем 1000

                    v_total_amount := v_total_amount + 1000;
                END LOOP;
        END LOOP;

    UPDATE bookings
    SET total_amount = v_total_amount
    WHERE book_ref = v_book_ref;

    RETURN v_book_ref;
END;
$$;
```
