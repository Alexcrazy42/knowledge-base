## 17.1

Постройте описание типа для хранения значений температуры. Температура может быть выражена в кельвинах и градусах Цельсия, Фаренгейта и Реомюра. Значения температуры не должны быть ниже 0 по шкале Кельвина. Сравните реализации в виде домена и в виде нового типа данных

```sql
-- Создаём базовый тип-перечисление для шкал
CREATE TYPE temp_scale AS ENUM ('C', 'K', 'F', 'Re');

-- Создаём кастомный тип (составной)
CREATE TYPE temperature AS (
                               value numeric(10, 2),
                               scale temp_scale
                           );

-- Функция для проверки корректности (абсолютный ноль)
CREATE OR REPLACE FUNCTION temperature_valid(t temperature)
    RETURNS boolean AS $$
BEGIN
    RETURN CASE t.scale
               WHEN 'K' THEN t.value >= 0
               WHEN 'C' THEN t.value >= -273.15
               WHEN 'F' THEN t.value >= -459.67
               WHEN 'Re' THEN t.value >= -218.52
        END;
END;
$$ LANGUAGE plpgsql IMMUTABLE;

-- Функция приведения к Кельвинам для сравнения
CREATE OR REPLACE FUNCTION to_kelvin(t temperature)
    RETURNS numeric AS $$
BEGIN
    RETURN CASE t.scale
               WHEN 'K' THEN t.value
               WHEN 'C' THEN t.value + 273.15
               WHEN 'F' THEN (t.value - 32) * 5/9 + 273.15
               WHEN 'Re' THEN t.value * 5/4 + 273.15
        END;
END;
$$ LANGUAGE plpgsql IMMUTABLE;

-- Оператор равенства
CREATE OR REPLACE FUNCTION temperature_eq(a temperature, b temperature)
    RETURNS boolean AS $$
BEGIN
    RETURN abs(to_kelvin(a) - to_kelvin(b)) < 0.0001;
END;
$$ LANGUAGE plpgsql IMMUTABLE;

CREATE OPERATOR = (
    LEFTARG = temperature,
    RIGHTARG = temperature,
    FUNCTION = temperature_eq,
    COMMUTATOR = =
    );

-- Оператор сравнения (меньше)
CREATE OR REPLACE FUNCTION temperature_lt(a temperature, b temperature)
    RETURNS boolean AS $$
BEGIN
    RETURN to_kelvin(a) < to_kelvin(b);
END;
$$ LANGUAGE plpgsql IMMUTABLE;

CREATE OPERATOR < (
    LEFTARG = temperature,
    RIGHTARG = temperature,
    FUNCTION = temperature_lt
    );

-- Функция для удобного создания значения температуры
CREATE OR REPLACE FUNCTION temp(value numeric, scale text)
    RETURNS temperature AS $$
DECLARE
    result temperature;
BEGIN
    result.value := value;
    result.scale := scale::temp_scale;

    IF NOT temperature_valid(result) THEN
        RAISE EXCEPTION 'Температура % % ниже абсолютного нуля', value, scale;
    END IF;

    RETURN result;
END;
$$ LANGUAGE plpgsql;
```

## 17.2

Опишите типы данных, необходимые для представления бронирования в виде одного значения составного типа. Информация о бронировании содержит основные атрибуты самого бронирования, список пассажиров и список беспосадочных перелетов.

```sql
CREATE TYPE segment_info AS (
    flight_id int,
    fare_condition text,
    price numeric (10, 2)
);

CREATE TYPE ticket_info AS (
    ticket_no text,
    passenger_id text,
    passenger_name text,
    outbound boolean,
    segments segment_info[]
);

CREATE TYPE bron_info AS (
    bron_ref char(3),
    bron_date timestamp with time zone,
    total_amount numeric(10, 2),
    tickets ticket_info[]
)
```
