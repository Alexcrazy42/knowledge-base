```sql
WITH ticket_segments_agg AS (
    SELECT
        t.ticket_no,
        t.book_ref,
        t.passenger_id,
        t.passenger_name,
        t.outbound,
        jsonb_agg(
                jsonb_build_object(

SELECT jsonb_build_object(
               'book_ref', b.book_ref,
               'book_date', b.book_date,
               'total_amount', b.total_amount,

                                       'passenger_name', tfa.passenger_name,
```

Создание триггера:

```
CREATE OR REPLACE FUNCTION check_airplane_range()
    RETURNS TRIGGER AS $$
DECLARE
    v_distance double precision;
    v_departureAirportPoint point;
    v_arrivalAirportPoint point;
    v_airplaneRange integer;
BEGIN
    
    SELECT coordinates INTO v_departureAirportPoint from bookings.airports_data 
        WHERE airport_code = NEW.departure_airport;
    
    SELECT coordinates INTO v_arrivalAirportPoint from bookings.airports_data 
        WHERE airport_code = NEW.arrival_airport;
    
    SELECT range INTO v_airplaneRange FROM bookings.airplanes_data WHERE airplane_code = NEW.airplane_code;
    
    v_distance := v_departureAirportPoint <@> v_arrivalAirportPoint * 1.60934;
    
    IF v_distance > v_airplaneRange  THEN
        RAISE EXCEPTION 'Самолет может лететь только на %, а вы хотите на %)', v_airplaneRange, v_distance;
    END IF;
    
    

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_airplane
    BEFORE INSERT OR UPDATE ON routes
    FOR EACH ROW
EXECUTE FUNCTION check_airplane_range();
```

## 8.7

Создайте в базе триггер, который не позволит выполнить операторы CREATE в ночное время

```
create function block_night_dml() returns trigger
    language plpgsql
as
$$
DECLARE
    current_hour INT := EXTRACT('hour' FROM CURRENT_TIME AT TIME ZONE 'Europe/Moscow');
BEGIN
    IF current_hour >= 23 OR current_hour < 6 THEN
        RAISE EXCEPTION 'Операции % запрещены в ночное время (23:00-06:00)',TG_OP; -- TG_OP содержит INSERT/UPDATE/DELETE
    END IF;

    RETURN NEW;
END;
$$;
```

Создание триггера на все таблицы:
```
DO $$
DECLARE
    r RECORD;
BEGIN
    FOR r IN (
        SELECT relname
        FROM pg_class
        WHERE relkind = 'r' 
          AND relnamespace = 'bookings'::regnamespace
          AND relname NOT LIKE 'pg_%' 
          AND relname NOT LIKE 'sql_%'
    ) LOOP
        EXECUTE format(
            'CREATE TRIGGER trg_block_night_%s
             BEFORE INSERT OR UPDATE OR DELETE ON %I
             FOR EACH ROW EXECUTE FUNCTION block_night_dml();',
            r.relname,
            r.relname
        );
    END LOOP;
END;
$$;
```

## 8.8

Создайте в базе вспомогательную таблицу и триггеры для аудита изменения рейсов. Изменение можно записывать в таблицу с тем же набором полей, а можно - в один JSON столбец (что позволяет избежать проблем при изменении структуры таблицы)

```sql
create table if not exists bookings.flights_changes
(
    flight_change bigint generated by default as identity
        primary key,
    flight_id     integer not null
        references bookings.flights,
    changes       jsonb
);

create function check_flight_changes() returns trigger
    language plpgsql
as
$$
BEGIN
    IF OLD.scheduled_departure IS DISTINCT FROM NEW.scheduled_departure
        OR OLD.scheduled_arrival IS DISTINCT FROM NEW.scheduled_arrival
        OR OLD.flight_id IS DISTINCT FROM NEW.flight_id
        OR OLD.status IS DISTINCT FROM NEW.status
    THEN
        INSERT INTO flights_changes
        (flight_id, changes)
        VALUES
            (NEW.flight_id,
             jsonb_build_object(
                     'status', NEW.status,
                     'newDeparture', NEW.scheduled_departure,
                     'newArrival', NEW.scheduled_arrival
             ));
    END IF;

    RETURN NEW;
END;
$$;

alter function check_flight_changes() owner to postgres;


create trigger trg_track_flight_changes
    after update
    on flights
    for each row
execute procedure check_flight_changes();


```

## 8.9

Создайте в базе событийный триггер, автоматически создающий для новых таблиц обычные триггеры для аудита изменений в этих таблицах


DML триггер - триггер, который срабатывает на изменение данных DML
Событийный триггер - триггер, срабатывающий на изменение DDL

```sql
CREATE TABLE audit_log (
                           id BIGSERIAL PRIMARY KEY,
                           table_name TEXT NOT NULL,
                           record_id TEXT NOT NULL,
                           operation TEXT NOT NULL,
                           old_data JSONB,
                           new_data JSONB,
                           changed_at TIMESTAMPTZ DEFAULT NOW(),
                           changed_by TEXT DEFAULT current_user,
                           changed_fields JSONB
);

CREATE INDEX idx_audit_table_record ON audit_log(table_name, record_id);
CREATE INDEX idx_audit_changed_at ON audit_log(changed_at);


CREATE OR REPLACE FUNCTION audit_trigger_function()
    RETURNS TRIGGER AS $$
DECLARE
    v_record_id TEXT;
    v_old_json JSONB;
    v_new_json JSONB;
    v_changed_fields JSONB := '{}'::JSONB;
    v_key TEXT;
    v_old_val TEXT;
    v_new_val TEXT;
BEGIN
    v_record_id := COALESCE(
            CASE WHEN NEW IS NOT NULL AND to_jsonb(NEW) ? 'id'
                     THEN (to_jsonb(NEW)->>'id')
                 WHEN OLD IS NOT NULL AND to_jsonb(OLD) ? 'id'
                     THEN (to_jsonb(OLD)->>'id')
                 ELSE NULL
                END,
            encode(sha256(COALESCE(to_jsonb(NEW)::text, to_jsonb(OLD)::text)::bytea), 'hex')
                   );

    IF TG_OP = 'INSERT' THEN
        v_new_json := to_jsonb(NEW);
        v_old_json := NULL;
    ELSIF TG_OP = 'DELETE' THEN
        v_old_json := to_jsonb(OLD);
        v_new_json := NULL;
    ELSIF TG_OP = 'UPDATE' THEN
        v_old_json := to_jsonb(OLD);
        v_new_json := to_jsonb(NEW);

        FOR v_key IN SELECT * FROM jsonb_object_keys(v_new_json) LOOP
                IF v_old_json->v_key IS DISTINCT FROM v_new_json->v_key THEN
                    v_old_val := COALESCE(v_old_json->>v_key, 'NULL');
                    v_new_val := COALESCE(v_new_json->>v_key, 'NULL');
                    v_changed_fields := v_changed_fields || jsonb_build_object(
                            v_key, jsonb_build_object('old', v_old_val, 'new', v_new_val)
                                                            );
                END IF;
            END LOOP;
    END IF;

    INSERT INTO audit_log (
        table_name,
        record_id,
        operation,
        old_data,
        new_data,
        changed_by,
        changed_fields
    ) VALUES (
                 TG_TABLE_NAME::TEXT,
                 v_record_id,
                 TG_OP,
                 v_old_json,
                 v_new_json,
                 current_user,
                 CASE WHEN TG_OP = 'UPDATE' THEN v_changed_fields ELSE NULL END
             );

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;


CREATE OR REPLACE FUNCTION create_audit_trigger_on_new_table()
    RETURNS event_trigger AS $$
DECLARE
    obj RECORD;
    table_schema TEXT;
    table_name TEXT;
    trigger_name TEXT;
BEGIN
    FOR obj IN SELECT * FROM pg_event_trigger_ddl_commands() LOOP
            IF obj.command_tag = 'CREATE TABLE' THEN
                table_schema := obj.object_identity::TEXT;
                -- Извлекаем имя таблицы (последняя часть)
                table_name := split_part(table_schema, '.', 2);
                table_schema := split_part(table_schema, '.', 1);

                trigger_name := 'trg_audit_' || table_name;

                -- Создаём триггер на новой таблице
                EXECUTE format('
                CREATE TRIGGER %I
                AFTER INSERT OR UPDATE OR DELETE ON %I.%I
                FOR EACH ROW
                EXECUTE FUNCTION audit_trigger_function()',
                               trigger_name, table_schema, table_name
                        );

                RAISE NOTICE 'Создан триггер аудита %.% на таблицу %.%',
                    table_schema, trigger_name, table_schema, table_name;
            END IF;
        END LOOP;
END;
$$ LANGUAGE plpgsql;


CREATE EVENT TRIGGER trg_create_audit_on_table_creation
    ON ddl_command_end
    WHEN TAG IN ('CREATE TABLE')
EXECUTE FUNCTION create_audit_trigger_on_new_table();
```
