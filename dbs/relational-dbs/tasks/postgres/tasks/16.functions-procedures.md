```sql
CREATE FUNCTION get_bron_info(p_book_ref char(6)) 
RETURNS TABLE(
    book_ref char(6),
    book_date timestamp with time zone,
    total_amount numeric(10, 2),
    passenger_count int
) 
AS $$
BEGIN
    RETURN QUERY
    SELECT 
        b.book_ref,
        b.book_date,
        b.total_amount,
        COUNT(t.book_ref) OVER (PARTITION BY b.book_ref)
    FROM bookings b
    JOIN tickets t on b.book_ref = t.book_ref

END;
$$ LANGUAGE plpgsql;
```

## 16.3

Напишите процедуру на языке PL/pgSQL, создающую новое бронирование в указанный день из заданного пункта отправления в заданный пункт назначения не более чем с двумя пересадками

```sql
CREATE OR REPLACE PROCEDURE create_booking_without_search(
    p_departure_airport char(3), 
    p_arrival_airport char(3),
    p_departure_date date,
    p_passenger_names text[]
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_flight_id integer;
    v_status text;
    v_airplane_code char(3);
    v_fare_condition text = 'Economy';
    v_passenger_name text;
    v_create_booking_param passenger_info[] := ARRAY[]::passenger_info[];
BEGIN
    
    SELECT 
        f.flight_id, r.airplane_code, f.status INTO v_flight_id, v_airplane_code, v_status
    FROM flights f
    JOIN routes r on f.route_no = r.route_no
    WHERE r.departure_airport = p_departure_airport
    AND r.arrival_airport = p_arrival_airport
    AND f.scheduled_departure >= p_departure_date
    AND f.scheduled_departure < p_departure_date + make_interval(days := 1)
    LIMIT 1;
    
    IF FOUND IS FALSE THEN
        RAISE EXCEPTION 'Перелет не найден!' USING ERRCODE = '50000';
    end if;
    
    IF v_status != 'Scheduled' 
    THEN
        RAISE EXCEPTION 'Рейс недоступен для бронирования!' USING ERRCODE = '50000';
    END IF;
    
    FOREACH v_passenger_name IN ARRAY p_passenger_names 
    LOOP
        v_create_booking_param := v_create_booking_param || ROW(
            v_passenger_name,
            ARRAY[
                ROW(v_flight_id, v_airplane_code, v_fare_condition)::airplane_fare_info
            ]
        )::passenger_info;
    END LOOP;


    PERFORM create_booking(v_create_booking_param);
END;
$$;
```

## 16.4

Напишите процедуру, добавляющую нового пассажира к указанному бронированию при условии, что на рейсе есть свободные места.

```sql
CREATE OR REPLACE PROCEDURE check_flight_availability(p_flight_id int, p_fare_condition text)
LANGUAGE plpgsql
AS $$
DECLARE
    v_airplane_code char(3);
    v_total_count INT;
    v_reserved_count INT;
BEGIN
    SELECT r.airplane_code INTO v_airplane_code 
    FROM flights f
    JOIN routes r on f.route_no = r.route_no
    WHERE f.flight_id = p_flight_id;
    
    SELECT COUNT(1) INTO v_reserved_count
    FROM segments s
    WHERE s.fare_conditions = p_fare_condition
    AND s.flight_id = p_flight_id;
    
    SELECT COUNT(1) INTO v_total_count
    FROM seats s
    WHERE s.airplane_code = v_airplane_code
    AND s.fare_conditions = p_fare_condition
    FOR UPDATE; -- блокируем race condition (блокируем все места самолета этого класса пока не создадим новый ticket)
    
    IF (v_total_count <= v_reserved_count) THEN
        RAISE EXCEPTION 'Нет доступных мест' USING ERRCODE = '50001';
    end if;
end;
$$;

CREATE OR REPLACE PROCEDURE add_passenger(p_book_ref char(3), p_passenger_name text)
LANGUAGE plpgsql
AS $$
DECLARE
    v_ticket_no text;
    v_flight_id int;
    v_fare_condition text;
    v_outbound bool;
BEGIN
    SELECT
        s.flight_id, s.fare_conditions, t.outbound INTO v_flight_id, v_fare_condition, v_outbound
    FROM tickets t
    JOIN segments s on t.ticket_no = s.ticket_no
    WHERE t.book_ref = p_book_ref
    LIMIT 1;
    
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Бронирование не найдено' USING ERRCODE = '50000';
    end if;

    CALL check_flight_availability(v_flight_id, v_fare_condition);

    v_ticket_no := generate_tocket_no();
    
    INSERT INTO tickets
    (ticket_no, book_ref, passenger_id, passenger_name, outbound)
    VALUES
    (v_ticket_no, p_book_ref, generate_passenger_id(), p_passenger_name, v_outbound);
    
    INSERT INTO segments
    (ticket_no, flight_id, fare_conditions, price) 
    VALUES 
    (v_ticket_no, v_flight_id, v_fare_condition, 1000);
    
    UPDATE bookings
    SET total_amount = total_amount + 1000
    WHERE book_ref = p_book_ref;
end;
$$

```

Проверьте состояние базы данных при нормальном выполнении, при нормальном выходе из обработки исключительной ситуации, при отсутствии обработки исключительной ситуации.

```sql
CREATE OR REPLACE procedure check_exception(throw bool)
LANGUAGE plpgsql
AS $$
DECLARE
    v_book_ref char(6) := '2EW1SQ';
BEGIN 
    UPDATE bookings SET total_amount = total_amount + 1 WHERE book_ref = v_book_ref;
    
    if throw THEN
        SELECT 1 / 0;
        
    end if;

    UPDATE bookings
    SET total_amount = total_amount + 2
    WHERE book_ref = v_book_ref;
    
end;
$$

SELECT * from bookings WHERE book_ref = '2EW1SQ';
CALL check_exception(true);

```
